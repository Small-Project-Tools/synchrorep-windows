cmake_minimum_required(VERSION 3.20)
project(synchrorep_windows VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(synchrorep_engine src/engine.cpp)
target_include_directories(synchrorep_engine PUBLIC include)
if(WIN32)
 add_executable(synchrorep WIN32 windows/main.cpp)
 target_link_libraries(synchrorep PRIVATE synchrorep_engine shell32 comctl32)
else()
 add_executable(synchrorep src/cli.cpp)
 target_link_libraries(synchrorep PRIVATE synchrorep_engine)
endif()
enable_testing()
add_executable(engine_tests tests/engine_tests.cpp)
target_link_libraries(engine_tests PRIVATE synchrorep_engine)
add_test(NAME engine_tests COMMAND engine_tests)
