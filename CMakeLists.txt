cmake_minimum_required(VERSION 3.20)
project(synchrorep_windows VERSION 0.12.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(SYNCHROREP_REMOTE "Build FTP/SFTP endpoints with libcurl" ON)
set(SYNCHROREP_ENGINE_SOURCES src/engine.cpp src/endpoint.cpp src/sha256.cpp src/profiles.cpp src/recovery_store.cpp src/recovery_catalog.cpp src/run_log.cpp src/recycle_bin.cpp src/mirror_monitor.cpp src/startup_profiles.cpp)
if(SYNCHROREP_REMOTE)
  find_package(CURL REQUIRED)
  list(APPEND SYNCHROREP_ENGINE_SOURCES src/remote_endpoint.cpp)
else()
  list(APPEND SYNCHROREP_ENGINE_SOURCES src/remote_stub.cpp)
endif()
add_library(synchrorep_engine ${SYNCHROREP_ENGINE_SOURCES})
target_include_directories(synchrorep_engine PUBLIC include)
if(SYNCHROREP_REMOTE)
  target_link_libraries(synchrorep_engine PUBLIC CURL::libcurl)
  target_compile_definitions(synchrorep_engine PUBLIC SYNCHROREP_REMOTE_ENABLED=1)
endif()
if(WIN32)
 target_link_libraries(synchrorep_engine PUBLIC ole32 uuid shell32)
 add_library(synchrorep_explorer_command SHARED windows/explorer_command.cpp)
 target_compile_definitions(synchrorep_explorer_command PRIVATE UNICODE _UNICODE)
 target_link_libraries(synchrorep_explorer_command PRIVATE shell32 ole32 uuid shlwapi)
 set_target_properties(synchrorep_explorer_command PROPERTIES OUTPUT_NAME "SynchrorepExplorer" PREFIX "")
 if(MINGW)
  target_link_options(synchrorep_explorer_command PRIVATE -static)
 endif()
 add_executable(synchrorep WIN32 windows/main.cpp windows/startup_registry.cpp)
 target_include_directories(synchrorep PRIVATE windows)
 add_executable(startup_registry_tests windows/startup_registry_tests.cpp windows/startup_registry.cpp)
 target_include_directories(startup_registry_tests PRIVATE windows)
 target_link_libraries(startup_registry_tests PRIVATE synchrorep_engine advapi32)
 if(MINGW)
  target_link_options(startup_registry_tests PRIVATE -static)
 endif()
 add_test(NAME startup_registry_tests COMMAND startup_registry_tests)
 add_executable(explorer_command_tests windows/explorer_command_tests.cpp)
 target_link_libraries(explorer_command_tests PRIVATE ole32 uuid)
 add_test(NAME explorer_command_tests COMMAND explorer_command_tests)
 add_dependencies(explorer_command_tests synchrorep_explorer_command)
 target_link_libraries(synchrorep PRIVATE synchrorep_engine shell32 comctl32 advapi32)
 if(MINGW)
  target_link_options(synchrorep PRIVATE -municode -static)
 endif()
else()
 add_executable(synchrorep src/cli.cpp)
 target_link_libraries(synchrorep PRIVATE synchrorep_engine)
endif()
enable_testing()
add_executable(windows_args_tests windows/windows_args_tests.cpp)
target_include_directories(windows_args_tests PRIVATE windows)
add_test(NAME windows_args_tests COMMAND windows_args_tests)
add_executable(recycle_bin_tests tests/recycle_bin_tests.cpp)
target_link_libraries(recycle_bin_tests PRIVATE synchrorep_engine)
add_test(NAME recycle_bin_tests COMMAND recycle_bin_tests)
add_executable(startup_tests tests/startup_tests.cpp)
target_link_libraries(startup_tests PRIVATE synchrorep_engine)
add_test(NAME startup_tests COMMAND startup_tests)
add_executable(profiles_tests tests/profiles_tests.cpp)
target_link_libraries(profiles_tests PRIVATE synchrorep_engine)
add_test(NAME profiles_tests COMMAND profiles_tests)
add_executable(sha256_tests tests/sha256_tests.cpp)
target_link_libraries(sha256_tests PRIVATE synchrorep_engine)
add_test(NAME sha256_tests COMMAND sha256_tests)
add_executable(recovery_catalog_tests tests/recovery_catalog_tests.cpp)
target_link_libraries(recovery_catalog_tests PRIVATE synchrorep_engine)
add_test(NAME recovery_catalog_tests COMMAND recovery_catalog_tests)
add_executable(recovery_tests tests/recovery_tests.cpp)
target_link_libraries(recovery_tests PRIVATE synchrorep_engine)
if(MINGW)
 target_link_options(recovery_tests PRIVATE -static)
endif()
add_test(NAME recovery_tests COMMAND recovery_tests)
add_executable(mirror_tests tests/mirror_tests.cpp)
target_link_libraries(mirror_tests PRIVATE synchrorep_engine)
add_test(NAME mirror_tests COMMAND mirror_tests)
add_executable(engine_tests tests/engine_tests.cpp)
target_link_libraries(engine_tests PRIVATE synchrorep_engine)
add_test(NAME engine_tests COMMAND engine_tests)
if(SYNCHROREP_REMOTE)
 add_executable(remote_tests tests/remote_tests.cpp)
 target_link_libraries(remote_tests PRIVATE synchrorep_engine)
 add_test(NAME remote_tests COMMAND remote_tests)
endif()
